cmake_minimum_required(VERSION 3.25)
project(FM6ExtraOptions LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Use MinGW-w64 GCC")
endif()
add_library(FM6ExtraOptions SHARED entry.cpp log.cpp ini.cpp patch.cpp pe.cpp hook.cpp live.cpp fov.cpp blur_amount.cpp)
target_compile_features(FM6ExtraOptions PRIVATE cxx_std_20)
target_compile_options(FM6ExtraOptions PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_options(FM6ExtraOptions PRIVATE -static -static-libgcc -static-libstdc++)
set_target_properties(FM6ExtraOptions PROPERTIES PREFIX "")
